        };
        
        state.inner_module_loading(self);

    fn cmr_link(&mut self) -> Result<(), Box<dyn Object>> {
        assert!(match self.get_status() {
            CyclicStatus::Unlinked | CyclicStatus::Linked | CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => true,
            _ => false,
        });

        let mut stack = List::new();
        let result = self.inner_module_linking(&mut stack, 0);

        if result.is_err() {
            for m in stack.iter_mut() {
                assert!(m.read().get_status() == CyclicStatus::Linking);

                m.write().set_status(CyclicStatus::Unlinked);
            }
            
            assert!(self.get_status() == CyclicStatus::Unlinked);
            return result.map(|_| ());
        }
        
        assert!(match self.get_status() {
            CyclicStatus::Linked | CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => true,
            _ => false,
        });
        assert!(stack.size() == 0);

        Ok(())
    }

    // TODO: returns a Promise object (NYI)
    fn cmr_evaluate(&mut self) -> ArcRw<dyn Object> {
        /*
         * TODO: if Agents are ever implemented we need to take a lock within
         * our agent to ensure that no evaluations happen simultaneously
         */
        // algorithm effectively has us override self. bleh. emulating
        let mut module: &mut (dyn CyclicModuleRec + '_) = match self.get_status() {
            CyclicStatus::Linked => CyclicModuleRec::as_mut_trait(self),
            CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => {
                if let Some(cycle_root) = self.get_cycle_root() {
                    if !addr_eq(Arc::as_ptr(&cycle_root), self.get_me().as_ptr()) {
                        &mut *cycle_root.write_arc()
                    }
                    else {
                        CyclicModuleRec::as_mut_trait(self)
                    }
                }
                else {
                    assert!(self.get_status() == CyclicStatus::Evaluated && self.get_evaluation_error().is_err());
                    CyclicModuleRec::as_mut_trait(self)
                }
            },
            _ => panic!("cmr_evaluate: invalid module status"),
        };

        if let Some(top_level_capability) = module.get_top_level_capability() {
            return top_level_capability.promise.clone();
        }
        
        //let stack = List::new();
        // TODO: FIXME: see comment in cmr_load_requested_modules
        let capability = PromiseCapability::new(JType::Undefined).unwrap();

        module.set_top_level_capability(Some(capability));
        let result = todo!(); // InnerModuleEvaluation
    }

pub trait ModuleRec {
    fn as_trait(&self) -> &dyn ModuleRec;

    fn as_mut_trait(&mut self) -> &mut dyn ModuleRec;
    /*

    fn is_cyclic(&self) -> Option<&dyn CyclicModuleRec>;

    fn is_cyclic_mut(&mut self) -> Option<&mut dyn CyclicModuleRec>;

    fn get_environment(&self) -> Option<ArcRw<ModuleEnv>>;

    fn inner_module_linking(&mut self, stack: &mut List<dyn CyclicModuleRec>, mut index: usize) -> Result<usize, Box<dyn Object>> {
        if let Some(module) = self.is_cyclic_mut() {
            match module.get_status() {
                CyclicStatus::Linking | CyclicStatus::Linked | CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => return Ok(index),
                CyclicStatus::Unlinked => {
                    module.set_status(CyclicStatus::Linking);
                    
                    let mut module_index = index;

                    module.set_dfs_ancestor_index(Some(index));
                    index += 1;
                    stack.append_arc(module.get_me().upgrade().unwrap());
                    for request in module.get_requested_modules().clone().iter() {
                        /*
                         * statement moved up here to prevent a double-borrow
                         * conflict with the deadlock prevention below
                         */
                        let mod_dfs_index = module.get_dfs_ancestor_index().unwrap();
                        let required_module: &mut (dyn ModuleRec + '_) = {
                            let ret = module.get_imported_module_mut(&*request.read());

                            if addr_eq(Arc::as_ptr(&ret), module.get_me().as_ptr()) {
                                module
                            }
                            else {
                                &mut *ret.write_arc()
                            }
                        };

                        index = required_module.inner_module_linking(stack, index)?;
                        if let Some(required_module) = required_module.is_cyclic_mut() {
                            match required_module.get_status() {
                                CyclicStatus::Linked | CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => (),
                                CyclicStatus::Linking => {
                                    assert!(stack.contains_weak(&required_module.get_me()).is_some());

                                    let req_dfs_index = required_module.get_dfs_ancestor_index().unwrap();

                                    module.set_dfs_ancestor_index(Some(mod_dfs_index.min(req_dfs_index)));
                                },
                                _ => panic!("inner_module_linking: BUG: invalid inner module status"),
                            }
                        }
                    }
                    module.initialize_environment()?;

                    assert!({
                        let mut once = 0;

                        for item in stack.iter() {
                            if once >= 2 {
                                break
                            }
                            if Arc::as_ptr(item) == module.get_me().as_ptr() {
                                once += 1;
                            }
                        }
                        once == 1
                    });
                    assert!(module.get_dfs_ancestor_index().unwrap() <= module_index);

                    if module.get_dfs_ancestor_index().unwrap() == module_index {
                        let mut done = false;

                        while !done {
                            /*
                             * could have used the infra stack struct but the
                             * spec refers to a normal List here. fortunately,
                             * we can deref into the Vec and pop() directly
                             * also, the spec doesn't account for the list
                             * being empty here, hence the unwrap() call.
                             */
                            let mut required_module = stack.pop().unwrap();

                            // modified from spec to prevent deadlocks
                            if addr_eq(Arc::as_ptr(&required_module), module.get_me().as_ptr()) {
                                module.set_status(CyclicStatus::Linked);
                                done = true;
                            }
                            else {
                                required_module.write().set_status(CyclicStatus::Linked);
                            }
                        }
                    }
                    return Ok(index);
                },
                _ => panic!("inner_module_linking: BUG: invalid module status"),
            }
        }
        self.link()?;
        Ok(index)
    }

    fn inner_module_evaluation(&mut self, stack: &mut List<dyn CyclicModuleRec>, mut index: usize) -> Result<usize, Box<dyn Object>> {
        if let Some(module) = self.is_cyclic_mut() {
            match module.get_status() {
                CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => {
                    module.get_evaluation_error().map(|_| index)
                },
                CyclicStatus::Evaluating => Ok(index),
                CyclicStatus::Linked => {
                    module.set_status(CyclicStatus::Evaluating);

                    let module_index = index;

                    module.set_dfs_ancestor_index(Some(index));
                    module.set_pending_async_dependencies(Some(0));
                    index += 1;
                    stack.append_arc(module.get_me().upgrade().unwrap());
                    for request in module.get_requested_modules().clone().iter() {
                        /*
                         * statement moved up here to prevent a double-borrow
                         * conflict with the deadlock prevention below
                         */
                        let mod_dfs_index = module.get_dfs_ancestor_index().clone().unwrap();
                        let mut required_module = {
                            let imp = module.get_imported_module_mut(&*request.read());

                            if addr_eq(Arc::as_ptr(&imp), module.get_me().as_ptr()) {
                                /*
                                 * note: effectively returns self,
                                 * but can't use self since it's
                                 * borrowed as module, then imp
                                 */
                                module.as_mut_abstract()
                            }
                            else {
                                &mut *imp.write_arc()
                            }
                        };

                        index = required_module.inner_module_evaluation(stack, index)?;
                        if let Some(mut required_module) = required_module.is_cyclic() {
                            match required_module.get_status() {
                                CyclicStatus::Evaluating => {
                                    assert!(stack.contains_weak(&required_module.get_me()).is_some());

                                    let req_dfs_index = required_module.get_dfs_ancestor_index().unwrap();
                                    
                                    module.set_dfs_ancestor_index(Some(mod_dfs_index.min(req_dfs_index)));
                                },
                                CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => {
                                    required_module = &*required_module.get_cycle_root().unwrap().read_recursive();
                                    match required_module.get_status() {
                                        CyclicStatus::EvaluatingAsync | CyclicStatus::Evaluated => {
                                            if let Err(e) = required_module.get_evaluation_error() {
                                                return Err(e);
                                            }
                                        },
                                        _ => panic!("inner_module_evaluation: BUG: invalid inner root module status"),
                                    }
                                },
                                _ => panic!("inner_module_evaluation: BUG: invalid inner module status"),
                            }
                            if let AsyncEvalOrder::Position(pos) = required_module.get_async_evaluation_order() {
                                module.set_pending_async_dependencies(Some(module.get_pending_async_dependencies().unwrap()+1));
                                required_module.mut_async_parent_modules().append_arc(module.get_me().upgrade().unwrap());
                            }
                        }
                    }
                    if module.get_pending_async_dependencies().unwrap() > 0 || module.has_tla() {
                        assert!(module.get_async_evaluation_order() == AsyncEvalOrder::Unset);

                        /*
                         * dammit, now i need to deal with agents. will start
                         * with a single global static record since right now
                         * everything will be single-threaded
                         */
                        module.set_async_evaluation_order(todo!()); // IncrementModuleAsyncEvaluationCount
                    }
                    todo!()
                },
                _ => panic!("inner_module_evaluation: BUG: invalid module status"),
            }
        }
        else {
            self.evaluate_module_sync()?;
            Ok(index)
        }
    }

    /*
     * using dyn Any instead of JAny for host_defined since it is
